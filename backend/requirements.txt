# Validation & settings
# The [email] extra pulls in email-validator, which is what backs EmailStr.
# Pydantic does not validate email addresses on its own.
# 2.13 rather than 2.10: google-genai needs pydantic >= 2.12.5.
pydantic[email]==2.13.5
pydantic-settings==2.7.0

# Auth
# PyJWT instead of python-jose: python-jose is unmaintained and its last
# release still carries unfixed CVEs. Both sign and verify the same tokens.
# The [crypto] extra is only needed for RS256/ES256. HS256 works without it,
# but it is here so switching to a key pair later is a one-line change.
pyjwt[crypto]==2.13.0

# pwdlib instead of passlib: passlib 1.7.4 reads bcrypt.__about__, which
# bcrypt removed in 4.1, so the pair only works if you pin bcrypt==4.0.1
# forever. pwdlib is the maintained replacement and needs no pin.
#
# The [argon2] extra, not [bcrypt], because PasswordHash.recommended() is
# Argon2. Argon2 is memory-hard and has no input length limit, while bcrypt
# 5.0 raises on passwords over 72 bytes instead of truncating them.
pwdlib[argon2]==0.3.1

pyotp==2.9.0

# File uploads
python-multipart==0.0.20

# LLM
# Google's Gemini SDK, used on the free tier. Only core/llm.py imports it, so
# a different provider can replace it without touching the rest of the app.
google-genai==2.25.0

# Testing
# In the same file as the app dependencies because there is one image and one
# local setup. Split into requirements-dev.txt if a production build is ever
# added, so pytest does not ship with the app.
pytest==9.1.1
pytest-asyncio==1.4.0
# httpx drives the app in-process through ASGITransport, so tests need no

httpx==0.28.1
